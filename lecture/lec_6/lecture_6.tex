\section{Systems that act}
% Content 1
\begin{frame}{Agents connect generated decisions to operations}
\small
\begin{columns}[c]
\column{.47\textwidth}
\begin{block}{Business uses}
An agent is a software component that uses an LLM to choose actions or next steps toward a goal, within application-defined limits.
\end{block}
LLM-generated text only becomes an action when application code executes an operation based on the text.
\par\medskip
Tool-using agents can select actions in response to observations \parencite{Yao2023ReAct}.
\column{.49\textwidth}
\centering
\begin{tikzpicture}[node distance=7mm,>=Latex,every node/.style={font=\small,align=center},box/.style={draw,rounded corners,text width=49mm,minimum height=9mm}]

\node[box,fill=orange!12,below=of q] (m) {model requests a query};
\node[box,fill=green!10,below=of m] (a) {application authorizes\\and executes a read};
\node[box,fill=blue!8,below=of a] (r) {model explains returned rows};
\draw[->,thick] (q)--(m);\draw[->,thick] (m)--(a);\draw[->,thick] (a)--(r);
\end{tikzpicture}
\visualdescription{A business question leads to a requested operation, an authorized read, and an evidence-based answer.}
\end{columns}
\end{frame}

% Content 2
\begin{frame}{Tools, workflows, and agents assign control differently}
\small
\begin{tabularx}{\linewidth}{@{}p{.17\linewidth}YY@{}}
\toprule
Concept & Who chooses the next step? & Sales example\\\midrule
Tool & Application invokes a defined operation & \texttt{run\_query(sql)} returns rows or an error.\\[.7em]
Workflow & Code follows predefined steps & Read schema $\to$ generate SQL $\to$ execute once $\to$ explain.\\[.7em]
Agent & Model selects some steps within code-enforced limits & Inspect schema, query, observe error, revise, or stop.\\\bottomrule
\end{tabularx}
\par\medskip
Autonomy concerns \emph{control flow}, not the number of tools or whether a framework is installed \parencite{Anthropic2024Agents}.
\visualdescription{The same query tool can support a fixed workflow or a model-directed sequence.}
\end{frame}

\begin{frame}{Give autonomy only where the next step is uncertain}
\small
\begin{columns}[c]
\column{.51\textwidth}

\column{.45\textwidth}
\begin{itemize}\itemsep.5em

\item Compare reliability, latency, usage, and oversight cost before increasing autonomy.
\end{itemize}

\section{Tool contracts}
% Content 4
\begin{frame}[fragile,t]{A tool contract defines a request, not permission}
\small
\lstset{aboveskip=3pt,belowskip=3pt}
\begin{columns}[T]
\column{.55\textwidth}
\textbf{Request contract}
\begin{lstlisting}[language=Python]
TOOLS = [{
 "type": "function",
 "name": "run_query",
 "description": "Read approved sales tables.",
 "parameters": {
   "type": "object",
   "properties": {"sql": {"type": "string"}},
   "required": ["sql"],
   "additionalProperties": False
 },
 "strict": True
}]
\end{lstlisting}
\visualdescription{TOOLS exposes run\_query to the model; slide 6 passes this contract into the model call.}
\column{.41\textwidth}
\textbf{Result contract}
\begin{lstlisting}[language=Python]
class QuerySuccess(TypedDict):
    ok: Literal[True]
    columns: list[str]
    rows: list[list]
    truncated: bool
    latency_ms: float

class QueryError(TypedDict):
    ok: Literal[False]
    error: str
    latency_ms: NotRequired[float]
QueryResult = (QuerySuccess
               | QueryError)
\end{lstlisting}
\visualdescription{Application-defined result types distinguish rows, errors, and truncation; error latency is optional. Typing imports omitted.}
\end{columns}
\par\smallskip\footnotesize The application defines and checks results; a truncated result is incomplete. \texttt{strict=True} constrains request arguments.
\slidesource{OpenAI (2026), \href{https://developers.openai.com/api/docs/guides/function-calling}{\emph{Function calling}}; result types describe the lecture's query executor}
\end{frame}

% Content 5
\begin{frame}[fragile]{Application code closes the function-calling loop}
\small
\lstset{aboveskip=3pt,belowskip=3pt}
\begin{columns}[T]
\column{.48\textwidth}
\textbf{Request a query using \texttt{TOOLS}}
\begin{lstlisting}[language=Python]
question = "How many orders are there?"
history = [{"role": "user",
            "content": question}]
response = client.responses.create(
    model=MODEL, instructions=POLICY,
    input=history, tools=TOOLS,
    parallel_tool_calls=False)
if response.status != "completed":
    raise RuntimeError("Incomplete output")
history.extend(response.output)
\end{lstlisting}
\visualdescription{TOOLS supplies the run\_query request contract from slide 5; the model receives it alongside the user's question.}
\column{.48\textwidth}
\textbf{Execute and return tool results}
\begin{lstlisting}[language=Python]
for call in response.output:
    if call.type != "function_call":
        continue
    record(call.name, call.arguments)
    result: QueryResult = dispatch(
        call.name, call.arguments)
    history.append({
        "type": "function_call_output",
        "call_id": call.call_id,
        "output": json.dumps(result)})
\end{lstlisting}
\visualdescription{After reading the TOOLS contract, the model requests run\_query with sql = ``SELECT COUNT(*) FROM orders''. The application executes the query and returns QueryResult with rows = [[9]].}
\end{columns}
\par\medskip\footnotesize
\begin{enumerate}\itemsep0pt\parsep0pt\topsep0pt
\item Call the model (left).
\item Validate and execute \texttt{run\_query}; append its \texttt{QueryResult} (right).
\item Call the model again with those results for its next step or final answer (left).
\end{enumerate}
\slidesource{OpenAI (2026), \href{https://developers.openai.com/api/docs/guides/function-calling}{\emph{Function calling}}. Code follows the lecture notebook}
\end{frame}

\section{Example tool: Text2SQL}
% Content 6
\begin{frame}{Text2SQL needs schema, row detail (grain), and definitions}
\small
\begin{columns}[c]
\column{.52\textwidth}
\centering
\begin{tikzpicture}[>=Latex,node distance=9mm,every node/.style={font=\scriptsize,align=left},b/.style={draw,rounded corners,text width=55mm,minimum height=13mm}]
\node[b,fill=blue!8] (c) {\textbf{customers}\\customer\_id, region};
\node[b,fill=green!10,below=of c] (o) {\textbf{orders}\\order\_id, customer\_id, date, status};
\node[b,fill=orange!12,below=of o] (i) {\textbf{order\_items}\\item\_id, order\_id, quantity, price, discount};
\draw[->] (c)--node[right,font=\tiny]{one to many} (o);
\draw[->] (o)--node[right,font=\tiny]{one to many} (i);
\end{tikzpicture}
\visualdescription{Grain means what each row represents: a customer, an order, or a line item. Joining orders to items produces line-item grain.}
\column{.44\textwidth}
\textbf{``Highest net sales last quarter''}
\begin{itemize}\itemsep.35em
\item Freeze ``today'' at 2026-10-15.
\item Last quarter: July--September 2026.
\item Include completed orders only.
\item Net line amount: quantity $\times$ unit price minus total line discount.
\item Missing discount: zero; missing region: Unknown. Money stored in cents.
\end{itemize}
\end{columns}
\par\smallskip\footnotesize Multi-table and cross-schema generalization are core Text2SQL challenges \parencite{Yu2018Spider}.
\end{frame}

% Content 7
\begin{frame}[fragile]{Executable SQL can still answer the wrong question}
\small
\begin{columns}[c]
\column{.60\textwidth}
\begin{lstlisting}[language=SQL,basicstyle=\ttfamily\scriptsize]
SELECT COALESCE(c.region, 'Unknown') AS region,
  SUM(i.quantity * i.unit_price_cents
      - COALESCE(i.discount_cents, 0))/100.0
      AS net_sales_usd
FROM customers c
JOIN orders o ON o.customer_id = c.customer_id
JOIN order_items i ON i.order_id = o.order_id
WHERE o.status = 'completed'
  AND o.order_date >= '2026-07-01'
  AND o.order_date <  '2026-10-01'
GROUP BY COALESCE(c.region, 'Unknown')
ORDER BY net_sales_usd DESC;
\end{lstlisting}
\visualdescription{The reference query joins at line-item grain and applies status, date, discount, and region rules.}
\column{.36\textwidth}
\begin{tabular}{lr}\toprule
Region & Net sales\\\midrule
East & \$420\\West & \$300\\South & \$90\\Unknown & \$40\\\bottomrule
\end{tabular}
\par\medskip\footnotesize Example reference results.
\setlength{\leftmargini}{1em}
\begin{itemize}\itemsep.35em
\item A query omitting the status filter still runs, but includes cancelled and pending sales.
\item ``Best customers'' needs a metric and period. Profit requires cost data that are absent.
\item The notebook uses the bounded loop to repair a seeded SQL column error.
\end{itemize}
\end{columns}
\end{frame}

\section[MCP]{External connectivity and MCP}
% Content 8
\begin{frame}{MCP standardizes tool discovery, contracts, and calls}
\footnotesize
MCP extends the contracts and dispatch across connected servers \parencite{MCP2025Tools}.
\begin{itemize}\footnotesize\itemsep0pt\parsep0pt
\item \textbf{Discover contracts:} \texttt{tools/list} returns tool descriptions, input schemas, or output schemas.
\item \textbf{Invoke tools:} \texttt{tools/call} sends a tool name and arguments.
\item \textbf{Receive results:} standard messages carry returned content and tool errors.
\end{itemize}
\centering
\includegraphics[width=.60\linewidth]{asset/model-context-protocol.pdf}
\visualdescription{The host uses clients to connect tool servers to data and services: stdio for local connections, Streamable HTTP for remote connections.}
\par\smallskip\scriptsize\raggedright
The host adapts contracts to the model API and enforces permissions. MCP does not grant authority to act.
\slidesource{Adapted from IBM (n.d.), \href{https://www.ibm.com/think/topics/model-context-protocol}{\emph{What is Model Context Protocol?}}; transport labels follow the MCP (2025) \href{https://modelcontextprotocol.io/specification/2025-11-25/basic/transports}{\emph{Transports} specification}}
\end{frame}

% Content 9
\begin{frame}[fragile]{MCP example: discover a calculator and invoke it}
\small
\begin{columns}[c]
\column{.52\textwidth}
\begin{lstlisting}[language=Python]
# Notebook host -> client -> local server
await session.initialize()
tools = await session.list_tools()
# Expose the approved contract to the model.
# Validate its request before invocation.
result = await session.call_tool(
    "revenue_growth",
    arguments={"previous": 1_200_000,
               "current": 1_500_000})
\end{lstlisting}
\visualdescription{Discovery supplies the contract; the model requests the operation and the host invokes the server.}
\column{.44\textwidth}
\begin{block}{Request $\to$ result $\to$ answer}
Previous revenue: \$1.2 million\\
Current revenue: \$1.5 million\\[.5em]
\texttt{growth\_percent: 25.0}\\[.5em]
$100\,(1.5-1.2)/1.2 = 25\%$
\end{block}
\footnotesize \textbf{Calculator example:} The notebook starts the calculator as a separate program on the same computer. They exchange MCP messages through the program's standard input and output (stdio). Only the notebook holds the model API key. Zero previous revenue is rejected.
\end{columns}
\slidesource{Model Context Protocol (2026), \href{https://github.com/modelcontextprotocol/python-sdk/tree/v1.x}{\emph{Python SDK v1}}. Full startup and cleanup appear in \texttt{mcp\_calculator.ipynb}.}
\end{frame}

\item Prior requests and observations
\item Successful tool evidence
\item Tool-call and error counters

% Content 11
\begin{frame}[fragile]{Error feedback is useful only inside stop rules}
\small
\begin{columns}[c]
\column{.51\textwidth}
\begin{lstlisting}[language=Python]
for turn in range(7):
    if calls >= 6 or errors >= 2:
        return "limit_reached"
    # Model may revise date using feedback.
    response = ask_model(history)
    if response_has_final_answer(response):
        return validate_answer(response)
    result = dispatch_checked_request(response)
    # Add tool results/errors to history.
    history += request_and_result(response, result)
    calls += 1
    errors += result_is_error_or_incomplete(result)
\end{lstlisting}
\visualdescription{Simplified control flow counts tool calls and errors and terminates unsuccessful runs explicitly.}
\column{.45\textwidth}
\begin{block}{A repair using tool feedback}
\texttt{lookup(date="Oct 1, 2026")}\\
$\downarrow$ tool expects \texttt{YYYY-MM-DD}\\
revise to \texttt{date="2026-10-01"}\\
$\downarrow$ validate and retry within limits
\end{block}
\footnotesize Illustrative validation error. Revise using supplied information; stop if errors persist or limits are reached.
\par\medskip API failure, incomplete output, limits, or truncation must never masquerade as completion.
\end{columns}
\end{frame}

\begin{frame}{Orchestration combines routing, sequencing, and feedback}
\small
\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash\hyphenpenalty=10000}p{.23\linewidth}YY@{}}

% Content 14
\begin{frame}[fragile]{Explicitly restrict what agents can do}
\small
\begin{columns}[c]
\column{.50\textwidth}
\begin{lstlisting}[language=Python]
# Model text cannot turn these controls off.
connect(uri + "?mode=ro", uri=True)
# PRAGMA sets SQLite's query-only mode.
connection.execute("PRAGMA query_only=ON")
connection.set_authorizer(allow_approved_reads)
connection.set_progress_handler(deadline, 100)
rows = cursor.fetchmany(51)  # 50-row cap
\end{lstlisting}
\visualdescription{The database and application enforce access and resource limits before results return to the model.}
\column{.46\textwidth}
\begin{itemize}\itemsep.35em
\item Block writes, private-table reads, attachment, and extension loading.
\item Flag truncated results and stop long queries cooperatively.
\item Minimize data sent to the provider and redact sensitive traces.
\item The Lecture 6 code example allows only three sales tables and named functions; it denies other operations.
\end{itemize}
\end{columns}
\par\medskip\footnotesize Prompt instructions don't guarantee safety. A production service needs stronger process isolation and identity-based authorization \parencite{SQLite2026Authorizer,SQLite2026Progress}.
\end{frame}

% Content 15
\begin{frame}[fragile]{Know which agent is acting and on whose behalf}
\small
\begin{columns}[c]
\column{.46\textwidth}
\textbf{Example deployment policy}
\begin{lstlisting}
agent_id: sales-query-01
owner: analytics_team
acting_for: user_42
allow: sales.read
access_expires_in: 10 minutes

agent_id: sales-report-01
allow: verified_results.read
deny: database.access
\end{lstlisting}
\visualdescription{Distinct agent identities receive different access. The query agent reads sales; the report agent receives verified results without database credentials.}
\column{.50\textwidth}
\begin{itemize}\footnotesize\itemsep.4em
\item \textbf{Identify the agent:} authenticate its software identity; a role name in a prompt is not proof \parencite{Booth2026AgentIdentity}.
\item \textbf{Link responsibility:} record the owner and the user or request behind each action.
\item \textbf{Protect credentials:} limit their access and lifetime; keep secrets out of prompts and agent handoffs.
\item \textbf{Check and track actions:} verify permissions on each tool call, log who did what, and provide a way to revoke access.
\end{itemize}
\end{columns}
\par\medskip\footnotesize Delegating a task must not expand access. Require human approval before consequential actions.
\slidesource{Booth et al., NIST (2026), \href{https://www.nist.gov/blogs/cybersecurity-insights/design-tenets-agent-identity-management-systems}{\emph{Design Tenets for Agent Identity Management Systems}}; \textcite{MCP2025Security}, \href{https://modelcontextprotocol.io/specification/2025-11-25/basic/security_best_practices}{\emph{Security Best Practices}}}

\begin{frame}{Trace observable actions from request to answer}

\begin{columns}[c]

Event & Evidence to retain\\\midrule

\begin{block}{Example diagnosis}

\end{block}
\footnotesize Record actions and external feedback. Do not require hidden chain-of-thought. Redact credentials and sensitive values before retaining traces.
\end{columns}
\end{frame}

% Content 18
\begin{frame}{Choose autonomy by value, evidence, and oversight cost}
\small
\begin{tabularx}{\linewidth}{@{}p{.21\linewidth}YY@{}}\toprule
Design & Useful when & What must be checked\\\midrule
Direct generation & Facts are already supplied & Faithfulness and missing information.\\[.5em]
Fixed workflow & The evidence path is known & Result correctness, calls, latency; errors stop.\\[.5em]
